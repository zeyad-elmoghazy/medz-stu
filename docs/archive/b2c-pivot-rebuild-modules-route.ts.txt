import { NextResponse } from 'next/server';
import { cookies } from 'next/headers';
import { createRouteHandlerClient } from '@/lib/supabase-server';
import type { Database } from '@/lib/supabase';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

type ModuleRow = {
  code: string;
  name: string;
  year_num: string;
  year_label: string;
  is_active: boolean;
};

type ChapterRow = {
  id: string;
  module_code: string;
  subject_id: string;
  slug: string;
  name: string;
  ordinal: number;
  question_count: number;
  published_count: number;
  flagged_count: number;
};

type SubjectRow = {
  id: string;
  slug: string;
  name: string;
};

/**
 * GET /api/modules
 *
 * Public-readable module catalog with per-subject chapters and
 * live counters. Relocated from /api/professor/modules during the
 * B2C pivot (015_b2c_pivot_rebuild.sql) — modules/chapters are no
 * longer professor-owned, and reads were always meant to be open
 * to any signed-in user. This version also reflects the
 * many-to-many module<->subject taxonomy: chapters are grouped
 * per (module, subject) pair rather than assuming one subject
 * per module.
 */
export async function GET() {
  const supabase = await createRouteHandlerClient<Database>({ cookies });

  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const client = supabase as unknown as {
    from: (t: string) => {
      select: (c: string) => {
        order: (c: string, o: { ascending: boolean }) => Promise<{ data: unknown[] | null; error: { message: string } | null }>;
      };
    };
  };

  const [modulesRes, subjectsRes, moduleSubjectsRes, chaptersRes] = await Promise.all([
    client.from('modules').select('code, name, year_num, year_label, is_active').order('code', { ascending: true }),
    client.from('subjects').select('id, slug, name').order('name', { ascending: true }),
    client.from('module_subjects').select('module_code, subject_id').order('module_code', { ascending: true }),
    client
      .from('chapters')
      .select('id, module_code, subject_id, slug, name, ordinal, question_count, published_count, flagged_count')
      .order('ordinal', { ascending: true }),
  ]);

  for (const res of [modulesRes, subjectsRes, moduleSubjectsRes, chaptersRes]) {
    if (res.error) {
      return NextResponse.json({ error: res.error.message }, { status: 500 });
    }
  }

  const modules = (modulesRes.data ?? []) as ModuleRow[];
  const subjects = (subjectsRes.data ?? []) as SubjectRow[];
  const moduleSubjects = (moduleSubjectsRes.data ?? []) as { module_code: string; subject_id: string }[];
  const chapters = (chaptersRes.data ?? []) as ChapterRow[];

  const subjectById = new Map(subjects.map((s) => [s.id, s]));

  const withSubjects = modules.map((m) => {
    const subjectIds = moduleSubjects
      .filter((ms) => ms.module_code === m.code)
      .map((ms) => ms.subject_id);

    const moduleSubjectList = subjectIds
      .map((sid) => {
        const subj = subjectById.get(sid);
        if (!subj) return null;
        return {
          id: subj.id,
          slug: subj.slug,
          name: subj.name,
          chapters: chapters
            .filter((c) => c.module_code === m.code && c.subject_id === sid)
            .map((c) => ({
              id: c.id,
              slug: c.slug,
              name: c.name,
              ordinal: c.ordinal,
              question_count: c.question_count,
              published_count: c.published_count,
              flagged_count: c.flagged_count,
            })),
        };
      })
      .filter((s): s is NonNullable<typeof s> => s !== null);

    // Flat `chapters` (merged across every subject on this module)
    // kept for callers that predate the multi-subject taxonomy and
    // just want "all chapters for this module code" — e.g. the
    // Histology-only subjects/[subjectId] browsing pages, pending
    // their rebuild against the full taxonomy (task: Custom Exam
    // builder / taxonomy-driven browsing UI).
    const flatChapters = chapters
      .filter((c) => c.module_code === m.code)
      .map((c) => ({
        id: c.id,
        slug: c.slug,
        name: c.name,
        ordinal: c.ordinal,
        question_count: c.question_count,
        published_count: c.published_count,
        flagged_count: c.flagged_count,
      }));

    return { ...m, subjects: moduleSubjectList, chapters: flatChapters };
  });

  return NextResponse.json(
    { modules: withSubjects },
    { headers: { 'Cache-Control': 'private, max-age=15' } }
  );
}
