'use client';

/**
 * Taxonomy Management (Part 2.4, section 2) — the 3-level
 * Module → Subject → Chapter editor. Subjects are a global
 * catalog joined to modules via module_subjects; a chapter
 * belongs to one (module, subject) pair per the 2026-08-07
 * schema correction in Part 4.2.
 *
 * Backend:
 *   GET/POST      /api/admin/modules
 *   PATCH         /api/admin/modules/[code]
 *   GET/POST      /api/admin/subjects            (global list, create)
 *   GET/POST/DELETE /api/admin/modules/[code]/subjects
 *   GET/POST      /api/admin/chapters?moduleCode=&subjectId=
 *   PATCH/DELETE  /api/admin/chapters/[id]
 */

import { useCallback, useEffect, useMemo, useState } from 'react';
import {
  Check,
  ChevronRight,
  Circle,
  Loader2,
  Pencil,
  Plus,
  Trash2,
  X,
} from 'lucide-react';
import { EmptyState, Panel, SectionHeader } from '../AdminShell';

type Module = {
  code: string;
  name: string;
  year_num: string;
  year_label: string;
  is_active: boolean;
  order_index: number;
};

type Subject = {
  id: string;
  slug: string;
  name: string;
  is_active: boolean;
};

type Chapter = {
  id: string;
  slug: string;
  name: string;
  order_index: number;
  question_count?: number;
};

export function TaxonomySection() {
  const [modules, setModules] = useState<Module[]>([]);
  const [subjectsCatalog, setSubjectsCatalog] = useState<Subject[]>([]);
  const [selectedModule, setSelectedModule] = useState<string | null>(null);
  const [moduleSubjects, setModuleSubjects] = useState<Subject[]>([]);
  const [selectedSubject, setSelectedSubject] = useState<string | null>(null);
  const [chapters, setChapters] = useState<Chapter[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const [showCreateModule, setShowCreateModule] = useState(false);
  const [showAssignSubject, setShowAssignSubject] = useState(false);
  const [showCreateChapter, setShowCreateChapter] = useState(false);

  const loadModules = useCallback(async () => {
    setLoading(true);
    try {
      const [mRes, sRes] = await Promise.all([
        fetch('/api/admin/modules', { credentials: 'include', cache: 'no-store' }),
        fetch('/api/admin/subjects', { credentials: 'include', cache: 'no-store' }),
      ]);
      if (!mRes.ok || !sRes.ok) {
        setError('Failed to load taxonomy');
        return;
      }
      const mBody = (await mRes.json()) as { modules: Module[] };
      const sBody = (await sRes.json()) as { subjects: Subject[] };
      setModules(mBody.modules ?? []);
      setSubjectsCatalog(sBody.subjects ?? []);
      setError(null);
    } catch {
      setError('Network error loading taxonomy');
    } finally {
      setLoading(false);
    }
  }, []);

  const loadModuleSubjects = useCallback(async (code: string) => {
    const res = await fetch(`/api/admin/modules/${code}/subjects`, {
      credentials: 'include',
      cache: 'no-store',
    });
    if (!res.ok) return;
    const body = (await res.json()) as { subjects: Subject[] };
    setModuleSubjects(body.subjects ?? []);
  }, []);

  const loadChapters = useCallback(async (moduleCode: string, subjectId: string) => {
    const url = new URL('/api/admin/chapters', window.location.origin);
    url.searchParams.set('moduleCode', moduleCode);
    url.searchParams.set('subjectId', subjectId);
    const res = await fetch(url.toString(), { credentials: 'include', cache: 'no-store' });
    if (!res.ok) return;
    const body = (await res.json()) as { chapters: Chapter[] };
    setChapters(body.chapters ?? []);
  }, []);

  useEffect(() => {
    void loadModules();
  }, [loadModules]);

  useEffect(() => {
    if (selectedModule) {
      setSelectedSubject(null);
      setChapters([]);
      void loadModuleSubjects(selectedModule);
    } else {
      setModuleSubjects([]);
    }
  }, [selectedModule, loadModuleSubjects]);

  useEffect(() => {
    if (selectedModule && selectedSubject) {
      void loadChapters(selectedModule, selectedSubject);
    }
  }, [selectedModule, selectedSubject, loadChapters]);

  const activeModule = useMemo(
    () => modules.find((m) => m.code === selectedModule) ?? null,
    [modules, selectedModule]
  );
  const activeSubject = useMemo(
    () => moduleSubjects.find((s) => s.id === selectedSubject) ?? null,
    [moduleSubjects, selectedSubject]
  );

  return (
    <div className="space-y-6">
      <SectionHeader
        eyebrow="Taxonomy"
        title="Modules · Subjects · Chapters"
        description="The three-level scope every question is tagged to. Chapters live at (module, subject) — the same subject can appear across modules with different chapters."
      />

      {error ? (
        <Panel>
          <div className="flex items-center justify-between text-sm text-rose-300">
            <span>{error}</span>
            <button
              type="button"
              onClick={() => void loadModules()}
              className="rounded-full px-3 py-1 text-xs text-white/80 hover:text-white"
              style={{ border: '1px solid #402039', backgroundColor: '#1a0d18' }}
            >
              Retry
            </button>
          </div>
        </Panel>
      ) : null}

      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
        <ColumnPanel
          title="Modules"
          count={modules.length}
          onAdd={() => setShowCreateModule(true)}
          addLabel="New module"
        >
          {loading ? (
            <ColumnLoading />
          ) : modules.length === 0 ? (
            <EmptyState title="No modules yet" description="Create Year 1 modules 101–104, 108 and Year 2 modules 205–208 to get started." />
          ) : (
            <ul className="space-y-1">
              {modules.map((m) => (
                <ColumnRow
                  key={m.code}
                  active={selectedModule === m.code}
                  onClick={() => setSelectedModule(m.code)}
                  primary={`M${m.code} · ${m.name}`}
                  secondary={`${m.year_label} · ${m.is_active ? 'live' : 'inactive'}`}
                  badge={m.is_active ? 'live' : 'draft'}
                />
              ))}
            </ul>
          )}
        </ColumnPanel>

        <ColumnPanel
          title={activeModule ? `Subjects in M${activeModule.code}` : 'Subjects'}
          count={moduleSubjects.length}
          onAdd={activeModule ? () => setShowAssignSubject(true) : undefined}
          addLabel="Assign subject"
          hint={activeModule ? undefined : 'Pick a module first'}
        >
          {!activeModule ? (
            <EmptyState title="Pick a module" description="Choose a module on the left to see the subjects assigned to it." />
          ) : moduleSubjects.length === 0 ? (
            <EmptyState
              title="No subjects assigned"
              description="Assign a subject from the global catalog to this module."
            />
          ) : (
            <ul className="space-y-1">
              {moduleSubjects.map((s) => (
                <ColumnRow
                  key={s.id}
                  active={selectedSubject === s.id}
                  onClick={() => setSelectedSubject(s.id)}
                  primary={s.name}
                  secondary={`slug: ${s.slug}`}
                  onRemove={async () => {
                    if (!confirm(`Unassign ${s.name} from M${activeModule.code}?`)) return;
                    const url = new URL(
                      `/api/admin/modules/${activeModule.code}/subjects`,
                      window.location.origin
                    );
                    url.searchParams.set('subjectId', s.id);
                    const res = await fetch(url.toString(), {
                      method: 'DELETE',
                      credentials: 'include',
                    });
                    if (res.ok) void loadModuleSubjects(activeModule.code);
                  }}
                />
              ))}
            </ul>
          )}
        </ColumnPanel>

        <ColumnPanel
          title={
            activeSubject && activeModule
              ? `Chapters in ${activeSubject.name} · M${activeModule.code}`
              : 'Chapters'
          }
          count={chapters.length}
          onAdd={activeModule && activeSubject ? () => setShowCreateChapter(true) : undefined}
          addLabel="New chapter"
          hint={activeModule && activeSubject ? undefined : 'Pick a subject first'}
        >
          {!activeSubject ? (
            <EmptyState title="Pick a subject" description="Choose a subject in the middle column to see its chapters within this module." />
          ) : chapters.length === 0 ? (
            <EmptyState title="No chapters yet" description="Add real chapter names (or placeholder ×3 to be renamed later, per Part 4.4)." />
          ) : (
            <ul className="space-y-1">
              {chapters.map((c) => (
                <ColumnRow
                  key={c.id}
                  primary={c.name}
                  secondary={`${c.question_count ?? 0} question(s) · order ${c.order_index}`}
                  onEdit={async () => {
                    const next = prompt('Rename chapter', c.name);
                    if (!next || next === c.name) return;
                    const res = await fetch(`/api/admin/chapters/${c.id}`, {
                      method: 'PATCH',
                      credentials: 'include',
                      headers: { 'content-type': 'application/json' },
                      body: JSON.stringify({ name: next }),
                    });
                    if (res.ok && selectedModule && selectedSubject) {
                      void loadChapters(selectedModule, selectedSubject);
                    }
                  }}
                  onRemove={async () => {
                    if ((c.question_count ?? 0) > 0) {
                      alert(
                        'This chapter still has questions. Re-tag or archive them first.'
                      );
                      return;
                    }
                    if (!confirm(`Delete chapter "${c.name}"?`)) return;
                    const res = await fetch(`/api/admin/chapters/${c.id}`, {
                      method: 'DELETE',
                      credentials: 'include',
                    });
                    if (res.ok && selectedModule && selectedSubject) {
                      void loadChapters(selectedModule, selectedSubject);
                    }
                  }}
                />
              ))}
            </ul>
          )}
        </ColumnPanel>
      </div>

      {showCreateModule ? (
        <CreateModuleModal
          onClose={() => setShowCreateModule(false)}
          onCreated={() => {
            setShowCreateModule(false);
            void loadModules();
          }}
        />
      ) : null}

      {showAssignSubject && activeModule ? (
        <AssignSubjectModal
          module={activeModule}
          catalog={subjectsCatalog}
          alreadyAssigned={new Set(moduleSubjects.map((s) => s.id))}
          onClose={() => setShowAssignSubject(false)}
          onDone={() => {
            setShowAssignSubject(false);
            void loadModuleSubjects(activeModule.code);
            void loadModules();
          }}
          onSubjectCatalogChanged={() => void loadModules()}
        />
      ) : null}

      {showCreateChapter && activeModule && activeSubject ? (
        <CreateChapterModal
          moduleCode={activeModule.code}
          subjectId={activeSubject.id}
          subjectName={activeSubject.name}
          onClose={() => setShowCreateChapter(false)}
          onCreated={() => {
            setShowCreateChapter(false);
            void loadChapters(activeModule.code, activeSubject.id);
          }}
        />
      ) : null}
    </div>
  );
}

function ColumnPanel({
  title,
  count,
  hint,
  onAdd,
  addLabel,
  children,
}: {
  title: string;
  count: number;
  hint?: string;
  onAdd?: () => void;
  addLabel?: string;
  children: React.ReactNode;
}) {
  return (
    <Panel padded={false}>
      <div
        className="flex items-center justify-between border-b px-4 py-3"
        style={{ borderColor: '#1E1E2E' }}
      >
        <div>
          <div className="text-sm font-semibold text-white">{title}</div>
          {hint ? <div className="text-[11px] text-white/40">{hint}</div> : null}
        </div>
        <div className="flex items-center gap-2">
          <span className="rounded-full px-2 py-0.5 text-[11px] text-white/60" style={{ backgroundColor: 'rgba(255,255,255,0.05)' }}>
            {count}
          </span>
          {onAdd ? (
            <button
              type="button"
              onClick={onAdd}
              className="inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-[11px] text-white/80 transition hover:text-white"
              style={{ border: '1px solid rgba(124,58,237,0.35)', backgroundColor: 'rgba(124,58,237,0.15)' }}
            >
              <Plus className="h-3 w-3" />
              {addLabel ?? 'Add'}
            </button>
          ) : null}
        </div>
      </div>
      <div className="max-h-[520px] overflow-auto p-3">{children}</div>
    </Panel>
  );
}

function ColumnRow({
  primary,
  secondary,
  badge,
  active,
  onClick,
  onEdit,
  onRemove,
}: {
  primary: string;
  secondary?: string;
  badge?: string;
  active?: boolean;
  onClick?: () => void;
  onEdit?: () => void;
  onRemove?: () => void;
}) {
  return (
    <li
      className="group flex items-center justify-between rounded-lg px-3 py-2 text-sm transition"
      style={
        active
          ? {
              backgroundColor: 'rgba(124, 58, 237, 0.18)',
              boxShadow: 'inset 0 0 0 1px rgba(159,103,255,0.35)',
              color: '#fff',
            }
          : { color: 'rgba(255,255,255,0.75)' }
      }
    >
      <button
        type="button"
        onClick={onClick}
        className="flex flex-1 items-center gap-2 text-left"
      >
        {onClick ? (
          <ChevronRight className="h-3.5 w-3.5 text-white/40 group-hover:text-white/80" />
        ) : (
          <Circle className="h-2 w-2 text-white/30" />
        )}
        <div className="flex-1">
          <div className="text-sm">{primary}</div>
          {secondary ? <div className="text-[11px] text-white/40">{secondary}</div> : null}
        </div>
      </button>
      {badge ? (
        <span
          className="rounded-full px-2 py-0.5 text-[10px] font-semibold uppercase"
          style={
            badge === 'live'
              ? {
                  backgroundColor: 'rgba(16,185,129,0.15)',
                  color: '#6EE7B7',
                  border: '1px solid rgba(16,185,129,0.35)',
                }
              : {
                  backgroundColor: 'rgba(255,255,255,0.06)',
                  color: 'rgba(255,255,255,0.6)',
                  border: '1px solid rgba(255,255,255,0.1)',
                }
          }
        >
          {badge}
        </span>
      ) : null}
      <div className="ml-2 flex items-center gap-1 opacity-0 transition group-hover:opacity-100">
        {onEdit ? (
          <button
            type="button"
            onClick={onEdit}
            className="rounded p-1 text-white/60 hover:text-white"
            title="Edit"
          >
            <Pencil className="h-3.5 w-3.5" />
          </button>
        ) : null}
        {onRemove ? (
          <button
            type="button"
            onClick={onRemove}
            className="rounded p-1 text-rose-300/80 hover:text-rose-300"
            title="Remove"
          >
            <Trash2 className="h-3.5 w-3.5" />
          </button>
        ) : null}
      </div>
    </li>
  );
}

function ColumnLoading() {
  return (
    <div className="flex items-center gap-2 px-3 py-6 text-sm text-white/50">
      <Loader2 className="h-4 w-4 animate-spin" /> Loading…
    </div>
  );
}

function CreateModuleModal({
  onClose,
  onCreated,
}: {
  onClose: () => void;
  onCreated: () => void;
}) {
  const [code, setCode] = useState('');
  const [name, setName] = useState('');
  const [yearNum, setYearNum] = useState('1');
  const [yearLabel, setYearLabel] = useState('Year 1');
  const [saving, setSaving] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  async function submit() {
    setErr(null);
    if (!/^\d{3}$/.test(code)) {
      setErr('Module code must be 3 digits.');
      return;
    }
    if (code[0] !== yearNum) {
      setErr(`Module code first digit (${code[0]}) should match year number (${yearNum}) — see Part 4.4.`);
      return;
    }
    setSaving(true);
    try {
      const res = await fetch('/api/admin/modules', {
        method: 'POST',
        credentials: 'include',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ code, name, year_num: yearNum, year_label: yearLabel }),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        setErr(body.error ?? `Failed (${res.status})`);
        return;
      }
      onCreated();
    } finally {
      setSaving(false);
    }
  }

  return (
    <Modal title="New module" onClose={onClose}>
      <div className="space-y-3">
        <Field label="Code (3 digits, e.g. 101)">
          <input
            value={code}
            onChange={(e) => setCode(e.target.value)}
            className="w-full rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
            placeholder="101"
            maxLength={3}
          />
        </Field>
        <Field label="Name">
          <input
            value={name}
            onChange={(e) => setName(e.target.value)}
            className="w-full rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
            placeholder="e.g. Introduction to Anatomy"
          />
        </Field>
        <div className="grid grid-cols-2 gap-3">
          <Field label="Year number">
            <input
              value={yearNum}
              onChange={(e) => setYearNum(e.target.value)}
              className="w-full rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
              style={{ border: '1px solid #1E1E2E' }}
              maxLength={2}
            />
          </Field>
          <Field label="Year label">
            <input
              value={yearLabel}
              onChange={(e) => setYearLabel(e.target.value)}
              className="w-full rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
              style={{ border: '1px solid #1E1E2E' }}
              placeholder="Year 1"
            />
          </Field>
        </div>
        {err ? <div className="text-xs text-rose-300">{err}</div> : null}
        <div className="flex justify-end gap-2 pt-2">
          <ModalButton onClick={onClose}>Cancel</ModalButton>
          <ModalButton primary onClick={submit} disabled={saving || !code || !name}>
            {saving ? 'Creating…' : 'Create module'}
          </ModalButton>
        </div>
      </div>
    </Modal>
  );
}

function AssignSubjectModal({
  module,
  catalog,
  alreadyAssigned,
  onClose,
  onDone,
  onSubjectCatalogChanged,
}: {
  module: Module;
  catalog: Subject[];
  alreadyAssigned: Set<string>;
  onClose: () => void;
  onDone: () => void;
  onSubjectCatalogChanged: () => void;
}) {
  const [mode, setMode] = useState<'existing' | 'new'>('existing');
  const [selected, setSelected] = useState<string | null>(null);
  const [newName, setNewName] = useState('');
  const [saving, setSaving] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  const available = useMemo(() => catalog.filter((s) => !alreadyAssigned.has(s.id)), [catalog, alreadyAssigned]);

  async function submit() {
    setErr(null);
    setSaving(true);
    try {
      if (mode === 'existing') {
        if (!selected) {
          setErr('Pick a subject to assign.');
          return;
        }
        const res = await fetch(`/api/admin/modules/${module.code}/subjects`, {
          method: 'POST',
          credentials: 'include',
          headers: { 'content-type': 'application/json' },
          body: JSON.stringify({ subjectId: selected }),
        });
        if (!res.ok) {
          setErr((await res.json().catch(() => ({}))).error ?? `Failed (${res.status})`);
          return;
        }
      } else {
        if (!newName.trim()) {
          setErr('Enter a subject name.');
          return;
        }
        const createRes = await fetch('/api/admin/subjects', {
          method: 'POST',
          credentials: 'include',
          headers: { 'content-type': 'application/json' },
          body: JSON.stringify({ name: newName.trim() }),
        });
        if (!createRes.ok) {
          setErr((await createRes.json().catch(() => ({}))).error ?? `Failed (${createRes.status})`);
          return;
        }
        const { subject } = (await createRes.json()) as { subject: Subject };
        const assignRes = await fetch(`/api/admin/modules/${module.code}/subjects`, {
          method: 'POST',
          credentials: 'include',
          headers: { 'content-type': 'application/json' },
          body: JSON.stringify({ subjectId: subject.id }),
        });
        if (!assignRes.ok) {
          setErr(`Created subject but assigning failed (${assignRes.status})`);
          onSubjectCatalogChanged();
          return;
        }
      }
      onDone();
    } finally {
      setSaving(false);
    }
  }

  return (
    <Modal title={`Assign a subject to M${module.code}`} onClose={onClose}>
      <div className="mb-3 flex gap-1 rounded-lg p-1" style={{ border: '1px solid #1E1E2E' }}>
        {(['existing', 'new'] as const).map((m) => (
          <button
            key={m}
            type="button"
            onClick={() => setMode(m)}
            className="flex-1 rounded-md px-3 py-1.5 text-xs capitalize"
            style={
              mode === m
                ? { backgroundColor: 'rgba(124,58,237,0.25)', color: '#fff', border: '1px solid rgba(159,103,255,0.35)' }
                : { color: 'rgba(255,255,255,0.6)' }
            }
          >
            {m === 'existing' ? 'Pick existing subject' : 'Create new subject'}
          </button>
        ))}
      </div>
      {mode === 'existing' ? (
        available.length === 0 ? (
          <div className="rounded-md px-3 py-4 text-center text-sm text-white/50" style={{ border: '1px dashed #1E1E2E' }}>
            Every subject in the global catalog is already assigned to this module. Create a new one instead.
          </div>
        ) : (
          <ul className="max-h-64 space-y-1 overflow-auto">
            {available.map((s) => (
              <li key={s.id}>
                <button
                  type="button"
                  onClick={() => setSelected(s.id)}
                  className="flex w-full items-center justify-between rounded-md px-3 py-2 text-left text-sm"
                  style={
                    selected === s.id
                      ? { backgroundColor: 'rgba(124,58,237,0.18)', color: '#fff', border: '1px solid rgba(159,103,255,0.35)' }
                      : { color: 'rgba(255,255,255,0.75)', border: '1px solid transparent' }
                  }
                >
                  <span>{s.name}</span>
                  {selected === s.id ? <Check className="h-4 w-4 text-violet-300" /> : null}
                </button>
              </li>
            ))}
          </ul>
        )
      ) : (
        <Field label="New subject name">
          <input
            value={newName}
            onChange={(e) => setNewName(e.target.value)}
            className="w-full rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
            placeholder="e.g. Embryology"
          />
        </Field>
      )}
      {err ? <div className="mt-3 text-xs text-rose-300">{err}</div> : null}
      <div className="mt-4 flex justify-end gap-2">
        <ModalButton onClick={onClose}>Cancel</ModalButton>
        <ModalButton primary onClick={submit} disabled={saving}>
          {saving ? 'Saving…' : 'Assign'}
        </ModalButton>
      </div>
    </Modal>
  );
}

function CreateChapterModal({
  moduleCode,
  subjectId,
  subjectName,
  onClose,
  onCreated,
}: {
  moduleCode: string;
  subjectId: string;
  subjectName: string;
  onClose: () => void;
  onCreated: () => void;
}) {
  const [name, setName] = useState('');
  const [orderIndex, setOrderIndex] = useState<number>(0);
  const [saving, setSaving] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  async function submit() {
    setErr(null);
    if (!name.trim()) {
      setErr('Enter a chapter name.');
      return;
    }
    setSaving(true);
    try {
      const res = await fetch('/api/admin/chapters', {
        method: 'POST',
        credentials: 'include',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({
          module_code: moduleCode,
          subject_id: subjectId,
          name: name.trim(),
          order_index: orderIndex,
        }),
      });
      if (!res.ok) {
        setErr((await res.json().catch(() => ({}))).error ?? `Failed (${res.status})`);
        return;
      }
      onCreated();
    } finally {
      setSaving(false);
    }
  }

  return (
    <Modal title={`New chapter · ${subjectName} · M${moduleCode}`} onClose={onClose}>
      <div className="space-y-3">
        <Field label="Chapter name">
          <input
            value={name}
            onChange={(e) => setName(e.target.value)}
            className="w-full rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
            placeholder="e.g. Cytology"
          />
        </Field>
        <Field label="Display order">
          <input
            type="number"
            value={orderIndex}
            onChange={(e) => setOrderIndex(Number(e.target.value))}
            className="w-24 rounded-md bg-black/30 px-3 py-2 text-sm text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
          />
        </Field>
        {err ? <div className="text-xs text-rose-300">{err}</div> : null}
        <div className="flex justify-end gap-2 pt-2">
          <ModalButton onClick={onClose}>Cancel</ModalButton>
          <ModalButton primary onClick={submit} disabled={saving}>
            {saving ? 'Creating…' : 'Create chapter'}
          </ModalButton>
        </div>
      </div>
    </Modal>
  );
}

function Modal({ title, onClose, children }: { title: string; onClose: () => void; children: React.ReactNode }) {
  return (
    <div className="fixed inset-0 z-40 flex items-center justify-center bg-black/70 p-4">
      <div
        className="w-full max-w-lg rounded-2xl p-6"
        style={{ backgroundColor: '#0F0F1A', border: '1px solid #1E1E2E' }}
      >
        <div className="mb-4 flex items-center justify-between">
          <h2 className="text-base font-semibold text-white">{title}</h2>
          <button type="button" onClick={onClose} className="rounded-full p-1 text-white/60 hover:text-white">
            <X className="h-4 w-4" />
          </button>
        </div>
        {children}
      </div>
    </div>
  );
}

function Field({ label, children }: { label: string; children: React.ReactNode }) {
  return (
    <label className="block">
      <div className="mb-1 text-[11px] uppercase tracking-[0.14em] text-white/50">{label}</div>
      {children}
    </label>
  );
}

function ModalButton({
  children,
  onClick,
  primary,
  disabled,
}: {
  children: React.ReactNode;
  onClick?: () => void;
  primary?: boolean;
  disabled?: boolean;
}) {
  return (
    <button
      type="button"
      onClick={onClick}
      disabled={disabled}
      className="rounded-md px-4 py-2 text-xs disabled:opacity-50"
      style={
        primary
          ? {
              backgroundColor: '#7C3AED',
              color: '#fff',
              boxShadow: '0 0 18px rgba(124,58,237,0.35)',
            }
          : { color: 'rgba(255,255,255,0.7)', border: '1px solid #1E1E2E' }
      }
    >
      {children}
    </button>
  );
}
