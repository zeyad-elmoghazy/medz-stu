'use client';

/**
 * Content Library / Review Queue (Part 2.4, sections 3 and 4).
 * Same table, driven by different default status filters:
 *
 *   Content Library: initialStatus = 'all' — every question,
 *     including published, directly editable.
 *   Review Queue:    initialStatus = 'draft,under_review' — AI
 *     extracted items awaiting first publish; the same edit / bulk
 *     controls apply but the default view is scoped down.
 *
 * Backend:
 *   GET /api/admin/questions?status=&moduleCode=&subjectId=&chapterId=&search=
 *   PATCH /api/admin/questions/[id]
 *   DELETE /api/admin/questions/[id]  (soft-delete → archived)
 *   GET /api/admin/questions/[id]/history
 *   POST /api/admin/questions/bulk    { ids[], action, chapterId? }
 */

import { useCallback, useEffect, useMemo, useState } from 'react';
import {
  Archive,
  Check,
  CheckCircle2,
  ChevronDown,
  Edit3,
  Eye,
  History,
  Loader2,
  RefreshCw,
  Search,
  Send,
  X,
} from 'lucide-react';
import { EmptyState, Panel, SectionHeader } from '../AdminShell';

type Choice = { id: string; text: string };

type QuestionRow = {
  id: number | string;
  question: string;
  choices: Choice[];
  correct_answer: string;
  explanation: string | null;
  reference: string | null;
  reference_book_id: string | null;
  reference_page: number | null;
  status: 'draft' | 'under_review' | 'published' | 'archived';
  difficulty: 'easy' | 'medium' | 'hard' | null;
  source: string | null;
  created_at: string;
  updated_at: string | null;
  chapter_id: string | null;
  chapters: {
    name: string | null;
    module_code: string | null;
    subject_id: string | null;
    subjects: { name: string | null } | null;
  } | null;
};

type Module = { code: string; name: string };
type Subject = { id: string; name: string };
type Chapter = { id: string; name: string };

const STATUS_TABS = [
  { id: 'all', label: 'All' },
  { id: 'draft,under_review', label: 'Review queue' },
  { id: 'published', label: 'Published' },
  { id: 'archived', label: 'Archived' },
] as const;

export function LibrarySection({
  initialStatus = 'all',
  title,
  description,
}: {
  initialStatus?: string;
  title: string;
  description: string;
}) {
  const [status, setStatus] = useState<string>(initialStatus);
  const [search, setSearch] = useState('');
  const [moduleCode, setModuleCode] = useState<string>('');
  const [subjectId, setSubjectId] = useState<string>('');
  const [chapterId, setChapterId] = useState<string>('');

  const [modules, setModules] = useState<Module[]>([]);
  const [moduleSubjects, setModuleSubjects] = useState<Subject[]>([]);
  const [chapters, setChapters] = useState<Chapter[]>([]);

  const [questions, setQuestions] = useState<QuestionRow[]>([]);
  const [total, setTotal] = useState(0);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const [selected, setSelected] = useState<Set<string>>(new Set());
  const [bulkBusy, setBulkBusy] = useState<null | 'publish' | 'archive'>(null);
  const [editing, setEditing] = useState<QuestionRow | null>(null);
  const [viewingHistory, setViewingHistory] = useState<QuestionRow | null>(null);

  // Load modules / subjects / chapters for filter dropdowns.
  useEffect(() => {
    (async () => {
      const res = await fetch('/api/admin/modules', { credentials: 'include', cache: 'no-store' });
      if (res.ok) {
        const body = (await res.json()) as { modules: Module[] };
        setModules(body.modules ?? []);
      }
    })();
  }, []);

  useEffect(() => {
    setSubjectId('');
    setChapterId('');
    setModuleSubjects([]);
    setChapters([]);
    if (!moduleCode) return;
    (async () => {
      const res = await fetch(`/api/admin/modules/${moduleCode}/subjects`, {
        credentials: 'include',
        cache: 'no-store',
      });
      if (res.ok) {
        const body = (await res.json()) as { subjects: Subject[] };
        setModuleSubjects(body.subjects ?? []);
      }
    })();
  }, [moduleCode]);

  useEffect(() => {
    setChapterId('');
    setChapters([]);
    if (!moduleCode || !subjectId) return;
    (async () => {
      const url = new URL('/api/admin/chapters', window.location.origin);
      url.searchParams.set('moduleCode', moduleCode);
      url.searchParams.set('subjectId', subjectId);
      const res = await fetch(url.toString(), { credentials: 'include', cache: 'no-store' });
      if (res.ok) {
        const body = (await res.json()) as { chapters: Chapter[] };
        setChapters(body.chapters ?? []);
      }
    })();
  }, [moduleCode, subjectId]);

  const load = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const url = new URL('/api/admin/questions', window.location.origin);
      if (status && status !== 'all') url.searchParams.set('status', status);
      if (search.trim()) url.searchParams.set('search', search.trim());
      if (moduleCode) url.searchParams.set('moduleCode', moduleCode);
      if (subjectId) url.searchParams.set('subjectId', subjectId);
      if (chapterId) url.searchParams.set('chapterId', chapterId);
      url.searchParams.set('limit', '100');
      const res = await fetch(url.toString(), { credentials: 'include', cache: 'no-store' });
      if (!res.ok) {
        setError(`Failed (${res.status})`);
        return;
      }
      const body = (await res.json()) as { questions: QuestionRow[]; total: number };
      setQuestions(body.questions ?? []);
      setTotal(body.total ?? 0);
      setSelected(new Set());
    } catch {
      setError('Network error');
    } finally {
      setLoading(false);
    }
  }, [status, search, moduleCode, subjectId, chapterId]);

  useEffect(() => {
    const t = window.setTimeout(() => void load(), 200);
    return () => window.clearTimeout(t);
  }, [load]);

  function toggleSelected(id: string) {
    setSelected((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  }
  function toggleAll() {
    setSelected((prev) => {
      if (prev.size === questions.length) return new Set();
      return new Set(questions.map((q) => String(q.id)));
    });
  }

  async function runBulk(action: 'publish' | 'archive') {
    if (selected.size === 0) return;
    setBulkBusy(action);
    try {
      const res = await fetch('/api/admin/questions/bulk', {
        method: 'POST',
        credentials: 'include',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ ids: Array.from(selected), action }),
      });
      if (!res.ok) {
        alert((await res.json().catch(() => ({}))).error ?? `Failed (${res.status})`);
        return;
      }
      void load();
    } finally {
      setBulkBusy(null);
    }
  }

  const activeChapterCount = questions.length;

  return (
    <div className="space-y-6">
      <SectionHeader
        eyebrow={initialStatus === 'draft,under_review' ? 'Review queue' : 'Content library'}
        title={title}
        description={description}
        right={
          <button
            type="button"
            onClick={() => void load()}
            className="inline-flex items-center gap-2 rounded-full px-3 py-2 text-xs text-white/70 transition hover:text-white"
            style={{ border: '1px solid #1E1E2E', backgroundColor: '#0F0F1A' }}
          >
            <RefreshCw className="h-3.5 w-3.5" />
            Refresh
          </button>
        }
      />

      <Panel padded={false}>
        <div
          className="grid gap-3 border-b px-5 py-4"
          style={{ borderColor: '#1E1E2E', gridTemplateColumns: 'minmax(0,1.5fr) repeat(3,minmax(0,1fr))' }}
        >
          <div
            className="flex items-center gap-2 rounded-md bg-black/30 px-3 py-2"
            style={{ border: '1px solid #1E1E2E' }}
          >
            <Search className="h-4 w-4 text-white/40" />
            <input
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              placeholder="Search question text"
              className="flex-1 bg-transparent text-sm text-white outline-none placeholder:text-white/30"
            />
          </div>
          <FilterSelect
            value={moduleCode}
            onChange={setModuleCode}
            options={[{ value: '', label: 'All modules' }, ...modules.map((m) => ({ value: m.code, label: `M${m.code} · ${m.name}` }))]}
          />
          <FilterSelect
            value={subjectId}
            onChange={setSubjectId}
            disabled={!moduleCode}
            options={[
              { value: '', label: moduleCode ? 'All subjects' : 'Pick a module first' },
              ...moduleSubjects.map((s) => ({ value: s.id, label: s.name })),
            ]}
          />
          <FilterSelect
            value={chapterId}
            onChange={setChapterId}
            disabled={!subjectId}
            options={[
              { value: '', label: subjectId ? 'All chapters' : 'Pick a subject first' },
              ...chapters.map((c) => ({ value: c.id, label: c.name })),
            ]}
          />
        </div>

        <div
          className="flex flex-wrap items-center gap-1 px-5 py-3"
          style={{ borderBottom: '1px solid #1E1E2E' }}
        >
          {STATUS_TABS.map((t) => (
            <button
              key={t.id}
              type="button"
              onClick={() => setStatus(t.id)}
              className="rounded-full px-3 py-1 text-xs transition"
              style={
                status === t.id
                  ? {
                      backgroundColor: 'rgba(124,58,237,0.22)',
                      color: '#fff',
                      border: '1px solid rgba(159,103,255,0.35)',
                    }
                  : { color: 'rgba(255,255,255,0.6)', border: '1px solid #1E1E2E' }
              }
            >
              {t.label}
            </button>
          ))}
          <span className="ml-auto text-[11px] text-white/50">
            Showing {activeChapterCount.toLocaleString()} of {total.toLocaleString()}
          </span>
        </div>

        {selected.size > 0 ? (
          <div
            className="flex flex-wrap items-center gap-2 px-5 py-3 text-xs text-white/80"
            style={{ borderBottom: '1px solid #1E1E2E', backgroundColor: 'rgba(124,58,237,0.08)' }}
          >
            <span>{selected.size} selected</span>
            <button
              type="button"
              onClick={() => runBulk('publish')}
              disabled={bulkBusy !== null}
              className="inline-flex items-center gap-1 rounded-md px-3 py-1 text-[11px] disabled:opacity-50"
              style={{
                backgroundColor: 'rgba(16,185,129,0.15)',
                color: '#6EE7B7',
                border: '1px solid rgba(16,185,129,0.35)',
              }}
            >
              <Send className="h-3 w-3" />
              {bulkBusy === 'publish' ? 'Publishing…' : 'Bulk publish'}
            </button>
            <button
              type="button"
              onClick={() => runBulk('archive')}
              disabled={bulkBusy !== null}
              className="inline-flex items-center gap-1 rounded-md px-3 py-1 text-[11px] disabled:opacity-50"
              style={{
                backgroundColor: 'rgba(255,255,255,0.05)',
                color: 'rgba(255,255,255,0.7)',
                border: '1px solid rgba(255,255,255,0.15)',
              }}
            >
              <Archive className="h-3 w-3" />
              {bulkBusy === 'archive' ? 'Archiving…' : 'Bulk archive'}
            </button>
            <button
              type="button"
              onClick={() => setSelected(new Set())}
              className="ml-auto text-white/60 hover:text-white"
            >
              Clear
            </button>
          </div>
        ) : null}

        {error ? (
          <div className="px-5 py-4 text-sm text-rose-300">{error}</div>
        ) : loading ? (
          <div className="flex items-center justify-center px-5 py-10 text-sm text-white/50">
            <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Loading…
          </div>
        ) : questions.length === 0 ? (
          <EmptyState
            title={initialStatus === 'draft,under_review' ? 'Review queue is clear' : 'No questions match'}
            description={
              initialStatus === 'draft,under_review'
                ? 'AI-extracted drafts will appear here once you upload a source file in the Content Upload tab.'
                : 'Try changing the filters, or use Manual MCQ Entry to add questions.'
            }
          />
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm">
              <thead
                className="text-[10px] uppercase tracking-[0.14em] text-white/50"
                style={{ borderBottom: '1px solid #1E1E2E' }}
              >
                <tr>
                  <th className="px-5 py-3">
                    <input
                      type="checkbox"
                      checked={selected.size > 0 && selected.size === questions.length}
                      onChange={toggleAll}
                      aria-label="Select all"
                    />
                  </th>
                  <th className="px-3 py-3 font-medium">Question</th>
                  <th className="px-3 py-3 font-medium">Scope</th>
                  <th className="px-3 py-3 font-medium">Status</th>
                  <th className="px-3 py-3 font-medium">Source</th>
                  <th className="px-5 py-3 font-medium text-right">Actions</th>
                </tr>
              </thead>
              <tbody>
                {questions.map((q) => (
                  <tr key={String(q.id)} style={{ borderBottom: '1px solid rgba(30,30,46,0.6)' }}>
                    <td className="px-5 py-3 align-top">
                      <input
                        type="checkbox"
                        checked={selected.has(String(q.id))}
                        onChange={() => toggleSelected(String(q.id))}
                      />
                    </td>
                    <td className="px-3 py-3 align-top">
                      <div className="line-clamp-2 max-w-md text-white/85">{q.question}</div>
                      <div className="mt-1 text-[11px] text-white/40">
                        {q.reference_page ? `p. ${q.reference_page}` : '—'} · {q.difficulty ?? 'unset'}
                      </div>
                    </td>
                    <td className="px-3 py-3 align-top text-white/70">
                      {q.chapters?.subjects?.name ?? '—'}
                      <div className="text-[11px] text-white/40">
                        M{q.chapters?.module_code ?? '?'} · {q.chapters?.name ?? '—'}
                      </div>
                    </td>
                    <td className="px-3 py-3 align-top">
                      <StatusBadge status={q.status} />
                    </td>
                    <td className="px-3 py-3 align-top text-[11px] text-white/50">
                      {q.source ?? '—'}
                    </td>
                    <td className="px-5 py-3 align-top text-right">
                      <div className="flex justify-end gap-1">
                        <IconBtn onClick={() => setEditing(q)} icon={Edit3} label="Edit" />
                        <IconBtn onClick={() => setViewingHistory(q)} icon={History} label="History" />
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </Panel>

      {editing ? (
        <EditQuestionModal
          question={editing}
          onClose={() => setEditing(null)}
          onSaved={() => {
            setEditing(null);
            void load();
          }}
        />
      ) : null}

      {viewingHistory ? (
        <HistoryModal question={viewingHistory} onClose={() => setViewingHistory(null)} />
      ) : null}
    </div>
  );
}

function FilterSelect({
  value,
  onChange,
  options,
  disabled,
}: {
  value: string;
  onChange: (v: string) => void;
  options: { value: string; label: string }[];
  disabled?: boolean;
}) {
  return (
    <div
      className="relative flex items-center rounded-md bg-black/30 px-3 py-2"
      style={{ border: '1px solid #1E1E2E', opacity: disabled ? 0.5 : 1 }}
    >
      <select
        value={value}
        disabled={disabled}
        onChange={(e) => onChange(e.target.value)}
        className="w-full appearance-none bg-transparent pr-6 text-sm text-white outline-none"
      >
        {options.map((o) => (
          <option key={o.value} value={o.value} style={{ color: '#000' }}>
            {o.label}
          </option>
        ))}
      </select>
      <ChevronDown className="pointer-events-none absolute right-2 h-4 w-4 text-white/40" />
    </div>
  );
}

function StatusBadge({ status }: { status: QuestionRow['status'] }) {
  const styles: Record<QuestionRow['status'], { bg: string; border: string; color: string }> = {
    published: { bg: 'rgba(16,185,129,0.15)', border: 'rgba(16,185,129,0.35)', color: '#6EE7B7' },
    under_review: { bg: 'rgba(245,158,11,0.15)', border: 'rgba(245,158,11,0.35)', color: '#FCD34D' },
    draft: { bg: 'rgba(148,163,184,0.15)', border: 'rgba(148,163,184,0.35)', color: '#CBD5E1' },
    archived: { bg: 'rgba(148,163,184,0.08)', border: 'rgba(148,163,184,0.2)', color: 'rgba(203,213,225,0.6)' },
  };
  const s = styles[status];
  return (
    <span
      className="rounded-full px-2 py-0.5 text-[10px] font-semibold uppercase"
      style={{ backgroundColor: s.bg, color: s.color, border: `1px solid ${s.border}` }}
    >
      {status.replace('_', ' ')}
    </span>
  );
}

function IconBtn({
  onClick,
  icon: Icon,
  label,
}: {
  onClick: () => void;
  icon: typeof Eye;
  label: string;
}) {
  return (
    <button
      type="button"
      onClick={onClick}
      title={label}
      className="grid h-7 w-7 place-items-center rounded-md text-white/70 transition hover:text-white"
      style={{ border: '1px solid #1E1E2E' }}
    >
      <Icon className="h-3.5 w-3.5" />
    </button>
  );
}

function EditQuestionModal({
  question,
  onClose,
  onSaved,
}: {
  question: QuestionRow;
  onClose: () => void;
  onSaved: () => void;
}) {
  const [text, setText] = useState(question.question);
  const [explanation, setExplanation] = useState(question.explanation ?? '');
  const [choices, setChoices] = useState<Choice[]>(question.choices);
  const [correct, setCorrect] = useState(question.correct_answer);
  const [status, setStatus] = useState(question.status);
  const [difficulty, setDifficulty] = useState<QuestionRow['difficulty']>(question.difficulty);
  const [saving, setSaving] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  async function save() {
    setErr(null);
    if (!choices.some((c) => c.id === correct)) {
      setErr('correctAnswer must match a choice id');
      return;
    }
    setSaving(true);
    try {
      const res = await fetch(`/api/admin/questions/${question.id}`, {
        method: 'PATCH',
        credentials: 'include',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({
          question: text,
          explanation,
          choices,
          correctAnswer: correct,
          status,
          difficulty: difficulty ?? 'medium',
        }),
      });
      if (!res.ok) {
        setErr((await res.json().catch(() => ({}))).error ?? `Failed (${res.status})`);
        return;
      }
      onSaved();
    } finally {
      setSaving(false);
    }
  }

  return (
    <ModalFrame title="Edit question" onClose={onClose} wide>
      <div className="space-y-4 text-sm">
        <Field label="Question text">
          <textarea
            value={text}
            onChange={(e) => setText(e.target.value)}
            rows={3}
            className="w-full rounded-md bg-black/30 px-3 py-2 text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
          />
        </Field>

        <div>
          <div className="mb-1 text-[11px] uppercase tracking-[0.14em] text-white/50">Choices</div>
          <div className="space-y-2">
            {choices.map((c, i) => (
              <div key={i} className="flex items-center gap-2">
                <button
                  type="button"
                  onClick={() => setCorrect(c.id)}
                  className="grid h-6 w-6 shrink-0 place-items-center rounded-full"
                  style={
                    correct === c.id
                      ? {
                          backgroundColor: 'rgba(16,185,129,0.2)',
                          border: '1px solid rgba(16,185,129,0.4)',
                          color: '#6EE7B7',
                        }
                      : {
                          backgroundColor: 'rgba(255,255,255,0.05)',
                          border: '1px solid rgba(255,255,255,0.15)',
                          color: 'rgba(255,255,255,0.5)',
                        }
                  }
                  title="Mark as correct answer"
                >
                  {correct === c.id ? <Check className="h-3 w-3" /> : c.id}
                </button>
                <input
                  value={c.text}
                  onChange={(e) => {
                    const next = [...choices];
                    next[i] = { ...c, text: e.target.value };
                    setChoices(next);
                  }}
                  className="flex-1 rounded-md bg-black/30 px-3 py-2 text-white outline-none"
                  style={{ border: '1px solid #1E1E2E' }}
                />
              </div>
            ))}
          </div>
        </div>

        <Field label="Explanation">
          <textarea
            value={explanation}
            onChange={(e) => setExplanation(e.target.value)}
            rows={3}
            className="w-full rounded-md bg-black/30 px-3 py-2 text-white outline-none"
            style={{ border: '1px solid #1E1E2E' }}
          />
        </Field>

        <div className="grid grid-cols-2 gap-3">
          <Field label="Status">
            <select
              value={status}
              onChange={(e) => setStatus(e.target.value as QuestionRow['status'])}
              className="w-full rounded-md bg-black/30 px-3 py-2 text-white outline-none"
              style={{ border: '1px solid #1E1E2E' }}
            >
              <option value="draft" style={{ color: '#000' }}>Draft</option>
              <option value="under_review" style={{ color: '#000' }}>Under review</option>
              <option value="published" style={{ color: '#000' }}>Published</option>
              <option value="archived" style={{ color: '#000' }}>Archived</option>
            </select>
          </Field>
          <Field label="Difficulty">
            <select
              value={difficulty ?? 'medium'}
              onChange={(e) => setDifficulty(e.target.value as QuestionRow['difficulty'])}
              className="w-full rounded-md bg-black/30 px-3 py-2 text-white outline-none"
              style={{ border: '1px solid #1E1E2E' }}
            >
              <option value="easy" style={{ color: '#000' }}>Easy</option>
              <option value="medium" style={{ color: '#000' }}>Medium</option>
              <option value="hard" style={{ color: '#000' }}>Hard</option>
            </select>
          </Field>
        </div>

        {err ? <div className="text-xs text-rose-300">{err}</div> : null}
        <div className="flex justify-end gap-2 pt-2">
          <ModalButton onClick={onClose}>Cancel</ModalButton>
          <ModalButton primary onClick={save} disabled={saving}>
            {saving ? 'Saving…' : 'Save changes'}
          </ModalButton>
        </div>
      </div>
    </ModalFrame>
  );
}

function HistoryModal({ question, onClose }: { question: QuestionRow; onClose: () => void }) {
  type HistoryRow = {
    id: string | number;
    action: string;
    summary: string | null;
    diff: Record<string, { before: unknown; after: unknown }> | null;
    created_at: string;
    profiles: { full_name: string | null; email: string | null } | null;
  };
  const [entries, setEntries] = useState<HistoryRow[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    (async () => {
      try {
        const res = await fetch(`/api/admin/questions/${question.id}/history`, {
          credentials: 'include',
          cache: 'no-store',
        });
        if (!res.ok) {
          setError(`Failed (${res.status})`);
          return;
        }
        const body = (await res.json()) as { history: HistoryRow[] };
        setEntries(body.history ?? []);
      } finally {
        setLoading(false);
      }
    })();
  }, [question.id]);

  return (
    <ModalFrame title="Edit history" onClose={onClose} wide>
      <p className="mb-3 line-clamp-2 text-xs text-white/50">{question.question}</p>
      {loading ? (
        <div className="flex items-center justify-center py-6 text-sm text-white/50">
          <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Loading…
        </div>
      ) : error ? (
        <div className="py-4 text-sm text-rose-300">{error}</div>
      ) : entries.length === 0 ? (
        <div className="py-6 text-center text-sm text-white/50">No history entries yet.</div>
      ) : (
        <ul className="space-y-3">
          {entries.map((e) => (
            <li
              key={String(e.id)}
              className="rounded-lg p-3 text-xs"
              style={{ border: '1px solid #1E1E2E', backgroundColor: 'rgba(255,255,255,0.02)' }}
            >
              <div className="flex items-center justify-between">
                <span className="text-white/80">
                  <CheckCircle2 className="mr-1 inline h-3 w-3 text-emerald-300" />
                  {e.summary ?? e.action}
                </span>
                <span className="text-white/40">
                  {new Date(e.created_at).toLocaleString()}
                </span>
              </div>
              <div className="mt-1 text-white/50">
                by {e.profiles?.full_name ?? e.profiles?.email ?? 'system'}
              </div>
              {e.diff ? (
                <div className="mt-2 space-y-1 rounded bg-black/30 p-2 font-mono text-[10px] text-white/70">
                  {Object.entries(e.diff).map(([field, change]) => (
                    <div key={field}>
                      <span className="text-violet-300">{field}</span>: {truncate(change.before)} → {truncate(change.after)}
                    </div>
                  ))}
                </div>
              ) : null}
            </li>
          ))}
        </ul>
      )}
    </ModalFrame>
  );
}

function truncate(v: unknown, max = 80): string {
  const s = typeof v === 'string' ? v : JSON.stringify(v);
  if (s == null) return 'null';
  return s.length > max ? s.slice(0, max) + '…' : s;
}

function ModalFrame({
  title,
  onClose,
  children,
  wide,
}: {
  title: string;
  onClose: () => void;
  children: React.ReactNode;
  wide?: boolean;
}) {
  return (
    <div className="fixed inset-0 z-40 flex items-center justify-center bg-black/70 p-4">
      <div
        className={`w-full ${wide ? 'max-w-2xl' : 'max-w-lg'} max-h-[90vh] overflow-y-auto rounded-2xl p-6`}
        style={{ backgroundColor: '#0F0F1A', border: '1px solid #1E1E2E' }}
      >
        <div className="mb-4 flex items-center justify-between">
          <h2 className="text-base font-semibold text-white">{title}</h2>
          <button type="button" onClick={onClose} className="rounded-full p-1 text-white/60 hover:text-white">
            <X className="h-4 w-4" />
          </button>
        </div>
        {children}
      </div>
    </div>
  );
}

function Field({ label, children }: { label: string; children: React.ReactNode }) {
  return (
    <label className="block">
      <div className="mb-1 text-[11px] uppercase tracking-[0.14em] text-white/50">{label}</div>
      {children}
    </label>
  );
}

function ModalButton({
  children,
  onClick,
  primary,
  disabled,
}: {
  children: React.ReactNode;
  onClick?: () => void;
  primary?: boolean;
  disabled?: boolean;
}) {
  return (
    <button
      type="button"
      onClick={onClick}
      disabled={disabled}
      className="rounded-md px-4 py-2 text-xs disabled:opacity-50"
      style={
        primary
          ? {
              backgroundColor: '#7C3AED',
              color: '#fff',
              boxShadow: '0 0 18px rgba(124,58,237,0.35)',
            }
          : { color: 'rgba(255,255,255,0.7)', border: '1px solid #1E1E2E' }
      }
    >
      {children}
    </button>
  );
}
